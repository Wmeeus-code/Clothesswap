async function pollResult(getUrl) { const statusText = document.getElementById('status-text'); const interval = setInterval(async () => { try { const res = await fetch("/api/tryon?getUrl=" + encodeURIComponent(getUrl)); const result = await res.json(); if (result.status === "succeeded") { clearInterval(interval); document.getElementById('loading-spinner').classList.add('hidden'); const resultImg = document.getElementById('result-img'); resultImg.src = result.output; resultImg.classList.remove('hidden'); statusText.innerText = "Pasbeurt voltooid!"; } else if (result.status === "failed") { clearInterval(interval); alert("Mislukt, probeer scherpere foto's."); document.getElementById('loading-spinner').classList.add('hidden'); document.getElementById('result-img').classList.remove('hidden'); } else { statusText.innerText = "AI past kleding aan... (" + result.status + ")"; } } catch (e) {} }, 3000); }
